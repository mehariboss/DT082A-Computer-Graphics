           function configureTexture( image ) {
              var texture = gl.createTexture();
              gl.bindTexture(gl.TEXTURE_2D, texture);

                // Flip the image's Y-axis to match WebGL's texture coordinate system
                gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, true);

                // Specify the texture image
                gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGB, gl.RGB, gl.UNSIGNED_BYTE, image);

                // Generate mipmaps for the texture & set the texture's minification/magnification filter to use mipmaps
                gl.generateMipmap(gl.TEXTURE_2D );
                gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.NEAREST_MIPMAP_LINEAR );
                gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.NEAREST );

                // Activate texture unit 0 and bind the texture to it
                gl.activeTexture(gl.TEXTURE0);

                // Pass the texture unit to the shader
                gl.uniform1i(gl.getUniformLocation(program, "texture"), 0);
           }

Applying textures

• Textures are applied during fragments shading by a sampler
• Samplers return a texture color from a texture object


Fragment shader:
           varying vec4 color;        //color from rasterizer
           varying vec2 texCoord;     //texture coordinate from rasterizer
           uniform sampler2D texture; //texture object from application

           void main() {
             gl_FragColor = color * texture2D( texture, texCoord );
           }
